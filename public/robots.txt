# See http://www.robotstxt.org/wc/norobots.html for documentation on how to use the robots.txt file
#
# To ban all spiders from the entire site uncomment the next two lines:
# User-Agent: *
# Disallow: /

# Follower lists are just pages of links to other people's profiles, which crawlers reach anyway, and they
# cost far more to render than they're worth indexing. Profiles themselves stay crawlable.
User-agent: *
Disallow: /users/*/followers
Disallow: /users/*/following
Disallow: /@*/followers
Disallow: /@*/following
